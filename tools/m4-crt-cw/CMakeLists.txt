cmake_minimum_required(VERSION 3.24)
project(MW34M4CRTCW LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4 OR NOT MSVC)
 message(FATAL_ERROR "Hardware CW research requires MSVC Windows x86")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(mw34_m4_cw_reference reference.cpp)
target_include_directories(mw34_m4_cw_reference PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../m4-crt")
target_compile_options(mw34_m4_cw_reference PRIVATE /W4 /WX /EHsc)
target_link_libraries(mw34_m4_cw_reference PRIVATE bcrypt)
