cmake_minimum_required(VERSION 3.24)
project(MW34X87Expand LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(MW34_X87_FIXTURE_KIND "0" CACHE STRING "Authored fixture: 0 spill64, 1 fistp32")
if(NOT MW34_X87_FIXTURE_KIND MATCHES "^[01]$")
    message(FATAL_ERROR "MW34_X87_FIXTURE_KIND must be 0 or 1")
endif()
if(NOT WIN32)
    message(FATAL_ERROR "The authored x86 expansion reference requires Windows")
endif()
if(CMAKE_SIZEOF_VOID_P EQUAL 4)
    enable_language(ASM_MASM)
    add_executable(mw34_x87_reference reference_worker.cpp reference.asm)
    target_compile_definitions(mw34_x87_reference PRIVATE MW34_X87_FIXTURE_KIND=${MW34_X87_FIXTURE_KIND})
    target_link_options(mw34_x87_reference PRIVATE /BASE:0x18000000 /DYNAMICBASE:NO /FIXED:NO)
    target_compile_options(mw34_x87_reference PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4>)
else()
    if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
        message(FATAL_ERROR "The x64 State bridge requires Clang's Remill-compatible type layout")
    endif()
    if(NOT MW34_LIFTED_OBJECT OR NOT MW34_REMILL_INCLUDE)
        message(FATAL_ERROR "x64 worker requires MW34_LIFTED_OBJECT and MW34_REMILL_INCLUDE")
    endif()
    add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../runtime" runtime)
    add_executable(mw34_x87_native native_worker.cpp "${MW34_LIFTED_OBJECT}")
    target_link_libraries(mw34_x87_native PRIVATE mw34_runtime)
    target_include_directories(mw34_x87_native SYSTEM PRIVATE "${MW34_REMILL_INCLUDE}")
    target_compile_definitions(mw34_x87_native PRIVATE MW34_X87_FIXTURE_KIND=${MW34_X87_FIXTURE_KIND}
        ADDRESS_SIZE_BITS=32 HAS_FEATURE_AVX=0 HAS_FEATURE_AVX512=0)
endif()
