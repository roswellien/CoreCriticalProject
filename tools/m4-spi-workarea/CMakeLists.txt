cmake_minimum_required(VERSION 3.24)
project(MW34SPIWorkarea LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32 OR NOT MW34_SPI_INCLUDE)
  message(FATAL_ERROR "Pinned Windows work-area query metadata required")
endif()
if(CMAKE_SIZEOF_VOID_P EQUAL 4)
  enable_language(ASM_MASM)
  add_executable(mw34_spi_original reference.cpp reference_x86.asm)
  target_include_directories(mw34_spi_original PRIVATE "${MW34_SPI_INCLUDE}")
  target_compile_options(mw34_spi_original PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/EHsc>)
  target_link_libraries(mw34_spi_original PRIVATE bcrypt user32)
  target_link_options(mw34_spi_original PRIVATE /BASE:0x18000000 /DYNAMICBASE:NO /FIXED:NO /LARGEADDRESSAWARE)
else()
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang" OR NOT MW34_REMILL_INCLUDE)
    message(FATAL_ERROR "Exact typed State and Windows x64 Clang required")
  endif()
  add_executable(mw34_spi_native native.cpp ../m4-startup/image_loader.cpp ../../runtime/memory/guest_memory.cpp)
  target_include_directories(mw34_spi_native PRIVATE "${MW34_SPI_INCLUDE}" ../m4-startup ../../runtime)
  target_include_directories(mw34_spi_native SYSTEM PRIVATE "${MW34_REMILL_INCLUDE}")
  target_compile_definitions(mw34_spi_native PRIVATE ADDRESS_SIZE_BITS=32 HAS_FEATURE_AVX=0 HAS_FEATURE_AVX512=0)
  target_compile_options(mw34_spi_native PRIVATE /EHsc)
  target_link_libraries(mw34_spi_native PRIVATE bcrypt user32)
endif()
