cmake_minimum_required(VERSION 3.26)
project(mw34_m4_assets LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(mw34_m4_assets_parser STATIC assets.cpp)
target_include_directories(mw34_m4_assets_parser PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(mw34_m4_assets_parser PUBLIC bcrypt)
if(MSVC)
    target_compile_options(mw34_m4_assets_parser PRIVATE /W4 /WX)
endif()
add_executable(mw34_m4_assets main.cpp)
target_link_libraries(mw34_m4_assets PRIVATE mw34_m4_assets_parser)
add_executable(mw34_m4_assets_tests ../../tests/m4-assets/assets_tests.cpp)
target_link_libraries(mw34_m4_assets_tests PRIVATE mw34_m4_assets_parser)
enable_testing()
add_test(NAME assets-authored-controls COMMAND mw34_m4_assets_tests)
