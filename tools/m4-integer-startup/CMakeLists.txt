cmake_minimum_required(VERSION 3.24)
project(MW34M4Slice LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT WIN32)
  message(FATAL_ERROR "M4 original private x86 reference requires Windows")
endif()
if(CMAKE_SIZEOF_VOID_P EQUAL 4)
  enable_language(ASM_MASM)
  add_executable(mw34_m4_reference reference_worker.cpp reference_x86.asm)
  target_compile_options(mw34_m4_reference PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4> $<$<COMPILE_LANGUAGE:CXX>:/EHsc>)
  target_link_libraries(mw34_m4_reference PRIVATE bcrypt)
  target_link_options(mw34_m4_reference PRIVATE /BASE:0x18000000 /DYNAMICBASE:NO /FIXED:NO /LARGEADDRESSAWARE)
else()
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang" OR NOT MW34_HOST_OBJECT OR NOT MW34_REMILL_INCLUDE)
    message(FATAL_ERROR "M4 native requires Clang, exact statically lifted object and pinned Remill State")
  endif()
  add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../runtime" runtime)
  add_executable(mw34_m4_native native_worker.cpp component.cpp "${MW34_HOST_OBJECT}")
  target_link_libraries(mw34_m4_native PRIVATE mw34_runtime bcrypt)
  target_include_directories(mw34_m4_native SYSTEM PRIVATE "${MW34_REMILL_INCLUDE}")
  target_compile_definitions(mw34_m4_native PRIVATE ADDRESS_SIZE_BITS=32 HAS_FEATURE_AVX=0 HAS_FEATURE_AVX512=0)
  target_compile_options(mw34_m4_native PRIVATE /EHsc)
  if(MW34_RUNTIME_SOURCE_DIR)
  add_executable(mw34_m4_runtime_component
    runtime_component_main.cpp runtime_adapter.cpp component.cpp "${MW34_HOST_OBJECT}"
    "${MW34_RUNTIME_SOURCE_DIR}/runtime.cpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/../m4-startup/image_loader.cpp")
  target_link_libraries(mw34_m4_runtime_component PRIVATE mw34_runtime bcrypt)
  target_include_directories(mw34_m4_runtime_component PRIVATE "${MW34_RUNTIME_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/../m4-startup")
  target_include_directories(mw34_m4_runtime_component SYSTEM PRIVATE "${MW34_REMILL_INCLUDE}")
  target_compile_definitions(mw34_m4_runtime_component PRIVATE ADDRESS_SIZE_BITS=32 HAS_FEATURE_AVX=0 HAS_FEATURE_AVX512=0)
  target_compile_options(mw34_m4_runtime_component PRIVATE /EHsc)
  endif()
endif()
