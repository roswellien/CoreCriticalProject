cmake_minimum_required(VERSION 3.24)
project(MW34M4CosineStatusRepair LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  message(FATAL_ERROR "Cosine repair requires Windows x64 Clang")
endif()
foreach(required MW34_HOST_OBJECTS MW34_PROBE_INCLUDE MW34_REMILL_INCLUDE)
  if(NOT DEFINED ${required} OR "${${required}}" STREQUAL "")
    message(FATAL_ERROR "Missing pinned repair dependency ${required}")
  endif()
endforeach()
foreach(mode fixed directed)
  if(mode STREQUAL "fixed")
    set(source ../m4-fp-transcendental/native_cosine.cpp)
  else()
    set(source ../m4-fp-transcendental/directed/native.cpp)
  endif()
  add_executable(mw34_m4_fp_cosine_${mode}_repair ${source} raw80_status.cpp ../m4-fp-store-repair/store_test.cpp ../../runtime/memory/guest_memory.cpp ${MW34_HOST_OBJECTS})
  target_compile_features(mw34_m4_fp_cosine_${mode}_repair PRIVATE cxx_std_20)
  target_include_directories(mw34_m4_fp_cosine_${mode}_repair PRIVATE ../../runtime "${MW34_PROBE_INCLUDE}")
  target_include_directories(mw34_m4_fp_cosine_${mode}_repair SYSTEM PRIVATE "${MW34_REMILL_INCLUDE}")
  target_compile_definitions(mw34_m4_fp_cosine_${mode}_repair PRIVATE ADDRESS_SIZE_BITS=32 HAS_FEATURE_AVX=0 HAS_FEATURE_AVX512=0)
  target_compile_options(mw34_m4_fp_cosine_${mode}_repair PRIVATE /EHsc /clang:-frounding-math)
  target_link_libraries(mw34_m4_fp_cosine_${mode}_repair PRIVATE bcrypt)
endforeach()
