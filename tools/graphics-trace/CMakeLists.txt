cmake_minimum_required(VERSION 3.24)
project(MW34GraphicsTrace LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "Graphics trace requires Win32 MSVC")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(mw34_graphics_trace SHARED trace.cpp exports.def)
target_link_libraries(mw34_graphics_trace PRIVATE bcrypt gdi32 user32)
add_executable(mw34_graphics_launcher ../in-game-hook/launcher.cpp)
target_link_libraries(mw34_graphics_launcher PRIVATE bcrypt)
foreach(target mw34_graphics_trace mw34_graphics_launcher)
    target_compile_definitions(${target} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_compile_options(${target} PRIVATE /W4 /WX /EHsc)
endforeach()
