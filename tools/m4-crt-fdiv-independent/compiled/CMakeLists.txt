cmake_minimum_required(VERSION 3.24)
project(MW34OwnFDIVClosure LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(WORKSPACE "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
add_executable(controls controls.cpp "${MW34_OWN_BOUNDARY}")
target_include_directories(controls PRIVATE "${WORKSPACE}/tools/m4-crt-fdiv" "${WORKSPACE}/tools/m4-runtime" "${WORKSPACE}/tools/m4-startup" "${WORKSPACE}/tools/m4-guest-cw" "${WORKSPACE}/runtime")
target_include_directories(controls SYSTEM PRIVATE "${WORKSPACE}/local/tools/remill-v6.0.1/remill-6.0.1/include")
target_compile_definitions(controls PRIVATE ADDRESS_SIZE_BITS=32 HAS_FEATURE_AVX=0 HAS_FEATURE_AVX512=0)
target_compile_options(controls PRIVATE /W4 /WX /EHsc /permissive- /clang:-Wno-cast-function-type-mismatch)
target_link_libraries(controls PRIVATE "${MW34_FDIV_LIBRARY}" bcrypt)
