cmake_minimum_required(VERSION 3.24)
project(MW34NumericMaxNumericalBoundary LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
string(REPLACE "/EHsc" "" CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS}")
get_filename_component(WORKSPACE "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
if(NOT MW34_NMX_OBJECT OR NOT MW34_NMX_SOURCE_DIR)
 message(FATAL_ERROR "Exact derived guest object and private derivative sources required")
endif()
add_executable(mw34_nmx_host_boundary "${MW34_NMX_SOURCE_DIR}/native.cpp" "${MW34_NMX_OBJECT}"
 "${WORKSPACE}/tools/m4-runtime/runtime.cpp" "${WORKSPACE}/tools/m4-startup/image_loader.cpp" "${WORKSPACE}/runtime/memory/guest_memory.cpp")
target_include_directories(mw34_nmx_host_boundary PRIVATE "${MW34_NMX_SOURCE_DIR}" "${WORKSPACE}/tools/m4-runtime" "${WORKSPACE}/tools/m4-startup" "${WORKSPACE}/runtime" "${WORKSPACE}/tools/m4-fp-numeric-max")
target_include_directories(mw34_nmx_host_boundary SYSTEM PRIVATE "${WORKSPACE}/local/tools/remill-v6.0.1/remill-6.0.1/include")
target_compile_options(mw34_nmx_host_boundary PRIVATE /W4 /WX /EHs /permissive-)
target_compile_definitions(mw34_nmx_host_boundary PRIVATE ADDRESS_SIZE_BITS=32 HAS_FEATURE_AVX=0 HAS_FEATURE_AVX512=0)
target_link_libraries(mw34_nmx_host_boundary PRIVATE bcrypt)
