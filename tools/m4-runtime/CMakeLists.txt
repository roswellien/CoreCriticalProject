cmake_minimum_required(VERSION 3.24)
project(MW34M4Runtime LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
get_filename_component(MW34_WORKSPACE "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_library(mw34_m4_runtime STATIC runtime.cpp
    "${MW34_WORKSPACE}/tools/m4-startup/image_loader.cpp"
    "${MW34_WORKSPACE}/runtime/memory/guest_memory.cpp")
target_include_directories(mw34_m4_runtime PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}"
    "${MW34_WORKSPACE}/tools/m4-startup" "${MW34_WORKSPACE}/runtime")
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "This admitted startup runtime currently targets Windows x64")
endif()
target_link_libraries(mw34_m4_runtime PUBLIC bcrypt)
add_executable(mw34_m4_runtime_host main.cpp)
target_link_libraries(mw34_m4_runtime_host PRIVATE mw34_m4_runtime)
include(CTest)
if(BUILD_TESTING)
    add_executable(mw34_m4_runtime_tests "${MW34_WORKSPACE}/tests/m4-runtime/runtime_tests.cpp")
    target_link_libraries(mw34_m4_runtime_tests PRIVATE mw34_m4_runtime)
    add_test(NAME m4_runtime_contract COMMAND mw34_m4_runtime_tests)
    add_executable(mw34_m4_service_continuation_tests "${MW34_WORKSPACE}/tests/m4-runtime/service_continuation_tests.cpp")
    target_link_libraries(mw34_m4_service_continuation_tests PRIVATE mw34_m4_runtime)
    add_test(NAME m4_service_continuation_contract COMMAND mw34_m4_service_continuation_tests "${MW34_WORKSPACE}/local/mw3-staged/Shared_DLLs/MSVCRT.DLL" "${MW34_WORKSPACE}/MW3/Mech3.exe")
endif()
foreach(target mw34_m4_runtime mw34_m4_runtime_host mw34_m4_runtime_tests mw34_m4_service_continuation_tests)
    if(TARGET ${target})
        if(MSVC)
            target_compile_options(${target} PRIVATE /W4 /WX /permissive-)
        else()
            target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
        endif()
    endif()
endforeach()
