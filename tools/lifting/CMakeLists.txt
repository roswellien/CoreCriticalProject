cmake_minimum_required(VERSION 3.24)
project(MW34FixtureWorker LANGUAGES CXX)
set(MW34_FIXTURE_KIND 0 CACHE STRING "Reviewed worker: 0=add, 1=branch, 2=memory, 3=shift, 4=private MW3 helper")
if(NOT MW34_FIXTURE_KIND MATCHES "^[01234]$")
    message(FATAL_ERROR "Unsupported worker kind")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT WIN32)
    message(FATAL_ERROR "The original x86 reference currently requires Windows")
endif()
if(CMAKE_SIZEOF_VOID_P EQUAL 4)
    enable_language(ASM_MASM)
    add_executable(mw34_reference reference_worker.cpp reference_x86.asm)
    # Keep the original game's 0x400000 allocation granularity region free.
    # Retain relocation metadata for hosts enforcing image relocation, while
    # opting out of ordinary per-image ASLR. Admission still checks every map.
    target_link_options(mw34_reference PRIVATE /BASE:0x18000000 /DYNAMICBASE:NO /FIXED:NO)
    target_link_libraries(mw34_reference PRIVATE bcrypt)
else()
    if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
        message(FATAL_ERROR "The x64 State bridge requires Clang's Remill-compatible type layout")
    endif()
    if(NOT MW34_LIFTED_OBJECT OR NOT MW34_REMILL_INCLUDE)
        message(FATAL_ERROR "x64 worker requires MW34_LIFTED_OBJECT and MW34_REMILL_INCLUDE")
    endif()
    add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../runtime" runtime)
    add_executable(mw34_lifted lifted_worker.cpp "${MW34_LIFTED_OBJECT}")
    target_link_libraries(mw34_lifted PRIVATE mw34_runtime bcrypt)
    target_include_directories(mw34_lifted SYSTEM PRIVATE "${MW34_REMILL_INCLUDE}")
    target_compile_definitions(mw34_lifted PRIVATE ADDRESS_SIZE_BITS=32 HAS_FEATURE_AVX=0 HAS_FEATURE_AVX512=0)
    target_compile_definitions(mw34_lifted PRIVATE MW34_FIXTURE_KIND=${MW34_FIXTURE_KIND})
endif()
