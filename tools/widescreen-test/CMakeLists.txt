cmake_minimum_required(VERSION 3.24)
project(MW34WidescreenTest LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "Widescreen test requires Win32 MSVC")
endif()
set(CMAKE_CXX_STANDARD 20)
add_library(mw34_fov_bootstrap SHARED bootstrap.cpp exports.def)
add_executable(mw34_fov_launcher ../in-game-hook/launcher.cpp)
foreach(target mw34_fov_bootstrap mw34_fov_launcher)
    target_compile_definitions(${target} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_compile_options(${target} PRIVATE /W4 /WX /EHsc)
    target_link_libraries(${target} PRIVATE bcrypt)
endforeach()
