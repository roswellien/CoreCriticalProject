cmake_minimum_required(VERSION 3.24)
project(MW34GuestCW LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(WORKSPACE "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_library(cw_memory STATIC "${WORKSPACE}/runtime/memory/guest_memory.cpp")
target_include_directories(cw_memory PUBLIC "${WORKSPACE}/runtime")
foreach(level O0 O2)
 add_library(mw34_guest_cw_${level} STATIC guest_cw.cpp)
 target_link_libraries(mw34_guest_cw_${level} PUBLIC cw_memory)
 target_include_directories(mw34_guest_cw_${level} PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
 target_include_directories(mw34_guest_cw_${level} SYSTEM PUBLIC "${WORKSPACE}/local/tools/remill-v6.0.1/remill-6.0.1/include")
 target_compile_definitions(mw34_guest_cw_${level} PUBLIC ADDRESS_SIZE_BITS=32 HAS_FEATURE_AVX=0 HAS_FEATURE_AVX512=0)
 target_compile_options(mw34_guest_cw_${level} PRIVATE /W4 /WX /permissive-)
 add_executable(mw34_guest_cw_tests_${level} "${WORKSPACE}/tests/m4-guest-cw/tests.cpp")
 target_link_libraries(mw34_guest_cw_tests_${level} PRIVATE mw34_guest_cw_${level})
 target_compile_options(mw34_guest_cw_tests_${level} PRIVATE /W4 /WX /permissive-)
 add_executable(mw34_guest_cw_replay_${level} native_replay.cpp)
 target_link_libraries(mw34_guest_cw_replay_${level} PRIVATE mw34_guest_cw_${level} bcrypt)
 target_include_directories(mw34_guest_cw_replay_${level} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../m4-crt")
 target_compile_options(mw34_guest_cw_replay_${level} PRIVATE /W4 /WX /permissive-)
 if(level STREQUAL "O0")
  target_compile_options(mw34_guest_cw_${level} PRIVATE /Od)
  target_compile_options(mw34_guest_cw_tests_${level} PRIVATE /Od)
  target_compile_options(mw34_guest_cw_replay_${level} PRIVATE /Od)
 endif()
endforeach()
enable_testing()
add_test(NAME guest_cw_O0 COMMAND mw34_guest_cw_tests_O0)
add_test(NAME guest_cw_O2 COMMAND mw34_guest_cw_tests_O2)
