cmake_minimum_required(VERSION 3.24)
project(MW34WindowedVulkanReplay LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "This first windowed replay uses Win32 presentation.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_path(VULKAN_HEADERS vulkan/vulkan_core.h REQUIRED)
find_program(GLSLANG_VALIDATOR NAMES glslangValidator glslang REQUIRED)
add_executable(mw34_vulkan_windowed viewer.cpp)
target_include_directories(mw34_vulkan_windowed PRIVATE "${VULKAN_HEADERS}")
target_link_libraries(mw34_vulkan_windowed PRIVATE user32)
if(MSVC)
    target_compile_options(mw34_vulkan_windowed PRIVATE /W4 /WX /EHsc)
endif()
foreach(stage vert frag)
    foreach(variant smooth flat)
        set(shader_name replay)
        set(shader_define "")
        if(variant STREQUAL flat)
            set(shader_name replay-flat)
            set(shader_define -DFLAT_SHADE=1)
        endif()
        set(shader_source "${CMAKE_CURRENT_SOURCE_DIR}/../vulkan-probe/replay.${stage}")
        set(shader_output "${CMAKE_CURRENT_BINARY_DIR}/${shader_name}.${stage}.spv")
        add_custom_command(OUTPUT "${shader_output}"
            COMMAND "${GLSLANG_VALIDATOR}" -V ${shader_define} "${shader_source}" -o "${shader_output}"
            DEPENDS "${shader_source}" VERBATIM)
        list(APPEND shader_outputs "${shader_output}")
    endforeach()
endforeach()
add_custom_target(windowed_shaders DEPENDS ${shader_outputs})
add_dependencies(mw34_vulkan_windowed windowed_shaders)
