cmake_minimum_required(VERSION 3.24)
project(MW34ControlCarryOriginal LANGUAGES CXX ASM_MASM)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4 OR NOT MW34_DERIVED)
  message(FATAL_ERROR "Isolated original carry requires Windows x86 and exact private derivatives")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(mw34_control_carry_original "${MW34_DERIVED}/original/reference.cpp" "${MW34_DERIVED}/original/reference_x86.asm")
target_include_directories(mw34_control_carry_original PRIVATE "${MW34_DERIVED}/original")
target_compile_options(mw34_control_carry_original PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4 /WX /permissive->)
target_link_libraries(mw34_control_carry_original PRIVATE bcrypt)
target_link_options(mw34_control_carry_original PRIVATE /BASE:0x60000000 /DYNAMICBASE:NO)
