cmake_minimum_required(VERSION 3.24)
project(MW34M4CosineDirected LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32 OR NOT MW34_PROBE_INCLUDE)
  message(FATAL_ERROR "Directed probe requires Windows and pinned probe metadata")
endif()
if(CMAKE_SIZEOF_VOID_P EQUAL 4)
  enable_language(ASM_MASM)
  add_executable(mw34_m4_fp_cosine_reference reference.cpp ../../m4-fp-startup/reference_x86.asm)
  target_include_directories(mw34_m4_fp_cosine_reference PRIVATE "${MW34_PROBE_INCLUDE}")
  target_compile_options(mw34_m4_fp_cosine_reference PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/EHsc>)
  target_link_libraries(mw34_m4_fp_cosine_reference PRIVATE bcrypt)
  target_link_options(mw34_m4_fp_cosine_reference PRIVATE /BASE:0x18000000 /DYNAMICBASE:NO /FIXED:NO /LARGEADDRESSAWARE)
else()
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang" OR NOT MW34_HOST_OBJECTS OR NOT MW34_REMILL_INCLUDE)
    message(FATAL_ERROR "Directed native requires exact static object and State")
  endif()
  add_executable(mw34_m4_fp_cosine_directed native.cpp ../raw80.cpp ../../../runtime/memory/guest_memory.cpp ${MW34_HOST_OBJECTS})
  target_include_directories(mw34_m4_fp_cosine_directed PRIVATE ../../../runtime "${MW34_PROBE_INCLUDE}")
  target_include_directories(mw34_m4_fp_cosine_directed SYSTEM PRIVATE "${MW34_REMILL_INCLUDE}")
  target_compile_definitions(mw34_m4_fp_cosine_directed PRIVATE ADDRESS_SIZE_BITS=32 HAS_FEATURE_AVX=0 HAS_FEATURE_AVX512=0)
  target_compile_options(mw34_m4_fp_cosine_directed PRIVATE /EHsc /clang:-frounding-math)
  target_link_libraries(mw34_m4_fp_cosine_directed PRIVATE bcrypt)
endif()
