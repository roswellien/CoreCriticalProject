cmake_minimum_required(VERSION 3.24)
project(MW34M4CosineFP LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  message(FATAL_ERROR "Cosine candidate requires Windows x64 Clang")
endif()
foreach(required MW34_HOST_OBJECTS MW34_PROBE_INCLUDE MW34_REMILL_INCLUDE)
  if(NOT DEFINED ${required} OR "${${required}}" STREQUAL "")
    message(FATAL_ERROR "Missing pinned candidate dependency ${required}")
  endif()
endforeach()
add_executable(mw34_m4_fp_cosine native_cosine.cpp raw80.cpp ../../runtime/memory/guest_memory.cpp ${MW34_HOST_OBJECTS})
target_compile_features(mw34_m4_fp_cosine PRIVATE cxx_std_20)
target_include_directories(mw34_m4_fp_cosine PRIVATE ../../runtime "${MW34_PROBE_INCLUDE}")
target_include_directories(mw34_m4_fp_cosine SYSTEM PRIVATE "${MW34_REMILL_INCLUDE}")
target_compile_definitions(mw34_m4_fp_cosine PRIVATE ADDRESS_SIZE_BITS=32 HAS_FEATURE_AVX=0 HAS_FEATURE_AVX512=0)
target_compile_options(mw34_m4_fp_cosine PRIVATE /EHsc /clang:-frounding-math)
target_link_libraries(mw34_m4_fp_cosine PRIVATE bcrypt)
