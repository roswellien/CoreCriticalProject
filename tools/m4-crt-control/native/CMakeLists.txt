cmake_minimum_required(VERSION 3.24)
project(MW34CRTControlNative LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(WORKSPACE "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
if(NOT MW34_CONTROL_INCLUDE OR NOT MW34_CONTROL_OBJECT)
 message(FATAL_ERROR "Exact compiled static provider object and metadata required")
endif()
add_executable(mw34_control_native ../native_worker.cpp "${WORKSPACE}/tools/m4-guest-cw/guest_cw.cpp" "${WORKSPACE}/runtime/memory/guest_memory.cpp" "${MW34_CONTROL_OBJECT}")
target_include_directories(mw34_control_native PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/.." "${MW34_CONTROL_INCLUDE}" "${WORKSPACE}/tools/m4-guest-cw" "${WORKSPACE}/runtime" "${WORKSPACE}/tools/m4-crt")
target_include_directories(mw34_control_native SYSTEM PRIVATE "${WORKSPACE}/local/tools/remill-v6.0.1/remill-6.0.1/include")
target_compile_definitions(mw34_control_native PRIVATE ADDRESS_SIZE_BITS=32 HAS_FEATURE_AVX=0 HAS_FEATURE_AVX512=0)
target_compile_options(mw34_control_native PRIVATE /W4 /WX /permissive-)
target_link_libraries(mw34_control_native PRIVATE bcrypt)
