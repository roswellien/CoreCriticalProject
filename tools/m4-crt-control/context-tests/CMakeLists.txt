cmake_minimum_required(VERSION 3.24)
project(MW34ControlContextTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(WORKSPACE "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
foreach(level O0 O2)
 add_executable(context_${level} "${WORKSPACE}/tests/m4-crt-control/context_tests.cpp" "${WORKSPACE}/tools/m4-guest-cw/guest_cw.cpp" "${WORKSPACE}/runtime/memory/guest_memory.cpp" "${MW34_CONTROL_OBJECT_DIR}/native-${level}.obj")
 target_include_directories(context_${level} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/.." "${MW34_CONTROL_INCLUDE}" "${WORKSPACE}/tools/m4-guest-cw" "${WORKSPACE}/runtime" "${WORKSPACE}/tools/m4-crt")
 target_include_directories(context_${level} SYSTEM PRIVATE "${WORKSPACE}/local/tools/remill-v6.0.1/remill-6.0.1/include")
 target_compile_definitions(context_${level} PRIVATE ADDRESS_SIZE_BITS=32 HAS_FEATURE_AVX=0 HAS_FEATURE_AVX512=0)
 target_compile_options(context_${level} PRIVATE /W4 /WX /permissive-)
 if(level STREQUAL "O0")
  target_compile_options(context_${level} PRIVATE /Od)
 endif()
 target_link_libraries(context_${level} PRIVATE bcrypt)
endforeach()
enable_testing()
foreach(level O0 O2)
 add_test(NAME context_${level} COMMAND context_${level} "${MW34_CONTROL_INCLUDE}/pack.bin")
endforeach()
