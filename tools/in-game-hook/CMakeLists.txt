cmake_minimum_required(VERSION 3.24)
project(MW34InGameHook LANGUAGES CXX ASM_MASM)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "This research hook requires a Win32 MSVC build")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(mw34_helper_hook SHARED helper_hook.cpp hook_x86.asm exports.def)
target_link_libraries(mw34_helper_hook PRIVATE bcrypt)
add_executable(mw34_hook_launcher launcher.cpp)
target_link_libraries(mw34_hook_launcher PRIVATE bcrypt)
add_executable(mw34_sample_context sample_context.cpp)
target_link_libraries(mw34_sample_context PRIVATE bcrypt psapi)
foreach(target mw34_helper_hook mw34_hook_launcher mw34_sample_context)
    target_compile_definitions(${target} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_compile_options(${target} PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4 /WX /EHsc>)
endforeach()
