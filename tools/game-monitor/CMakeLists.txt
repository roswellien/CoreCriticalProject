cmake_minimum_required(VERSION 3.24)
project(MW34MonitorProbe LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "The authored Windows diagnostic probe requires MSVC")
endif()
add_executable(monitor_probe probe.cpp)
target_compile_features(monitor_probe PRIVATE cxx_std_20)
target_compile_definitions(monitor_probe PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(monitor_probe PRIVATE /W4 /WX /EHsc)
