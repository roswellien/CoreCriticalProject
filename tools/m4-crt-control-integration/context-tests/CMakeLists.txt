cmake_minimum_required(VERSION 3.24)
project(MW34ConnectedControlCausal LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(WORKSPACE "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
if(NOT MW34_CONNECTED_INCLUDE OR NOT MW34_CONNECTED_OBJECT_DIR)
 message(FATAL_ERROR "Exact connected build dependencies required")
endif()
enable_testing()
foreach(level O0 O2)
 file(SHA256 "${MW34_CONNECTED_OBJECT_DIR}/native-${level}.obj" artifact)
 add_executable(connected_context_${level} "${WORKSPACE}/tests/m4-crt-control-integration/context_tests.cpp"
  "${WORKSPACE}/tools/m4-runtime/runtime.cpp" "${WORKSPACE}/tools/m4-startup/image_loader.cpp"
  "${WORKSPACE}/tools/m4-guest-cw/guest_cw.cpp" "${WORKSPACE}/runtime/memory/guest_memory.cpp" "${MW34_CONNECTED_OBJECT_DIR}/native-${level}.obj")
 target_include_directories(connected_context_${level} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/.." "${MW34_CONNECTED_INCLUDE}" "${WORKSPACE}/tools/m4-runtime" "${WORKSPACE}/tools/m4-startup" "${WORKSPACE}/tools/m4-guest-cw" "${WORKSPACE}/runtime" "${WORKSPACE}/tools/m4-crt")
 target_include_directories(connected_context_${level} SYSTEM PRIVATE "${WORKSPACE}/local/tools/remill-v6.0.1/remill-6.0.1/include")
 target_compile_definitions(connected_context_${level} PRIVATE ADDRESS_SIZE_BITS=32 HAS_FEATURE_AVX=0 HAS_FEATURE_AVX512=0 MW34_CONNECTED_ARTIFACT_SHA="${artifact}")
 target_compile_options(connected_context_${level} PRIVATE /W4 /WX /permissive-)
 if(level STREQUAL O0)
  target_compile_options(connected_context_${level} PRIVATE /Od)
 else()
  target_compile_options(connected_context_${level} PRIVATE /O2)
 endif()
 target_link_libraries(connected_context_${level} PRIVATE bcrypt)
 add_test(NAME connected_context_${level} COMMAND connected_context_${level} "${MW34_CONNECTED_INCLUDE}/pack.bin" "${WORKSPACE}/local/mw3-staged/Shared_DLLs/MSVCRT.DLL" "${WORKSPACE}/MW3/Mech3.exe")
endforeach()
