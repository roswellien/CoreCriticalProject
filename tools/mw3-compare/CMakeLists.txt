cmake_minimum_required(VERSION 3.24)
project(MW34Compare LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "The original-side comparison hook requires a Win32 MSVC build")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(mw34_compare_hook SHARED compare_hook.cpp exports.def)
target_link_libraries(mw34_compare_hook PRIVATE dxguid user32 gdi32 advapi32 vfw32)
add_executable(mw34_compare_launcher compare_launcher.cpp)
target_link_libraries(mw34_compare_launcher PRIVATE bcrypt user32)
foreach(target mw34_compare_hook mw34_compare_launcher)
    target_compile_definitions(${target} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX DIRECTINPUT_VERSION=0x0500 _CRT_SECURE_NO_WARNINGS)
    target_compile_options(${target} PRIVATE /W4 /EHsc /MT /Oy-)
endforeach()
target_compile_definitions(mw34_compare_launcher PRIVATE UNICODE _UNICODE)
