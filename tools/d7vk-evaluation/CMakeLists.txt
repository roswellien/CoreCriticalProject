cmake_minimum_required(VERSION 3.24)
project(MW34D7VKEvaluation LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "This isolated evaluation requires Windows and MSVC.")
endif()
add_executable(mw34_d7vk_fixture fixture.cpp)
target_compile_features(mw34_d7vk_fixture PRIVATE cxx_std_20)
target_compile_definitions(mw34_d7vk_fixture PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(mw34_d7vk_fixture PRIVATE /W4 /WX /EHsc)
target_link_libraries(mw34_d7vk_fixture PRIVATE user32 gdi32 dxguid)
