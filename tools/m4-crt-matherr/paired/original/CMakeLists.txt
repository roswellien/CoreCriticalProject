cmake_minimum_required(VERSION 3.24)
project(MW34MatherrOriginal LANGUAGES CXX ASM_MASM)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(WORKSPACE "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
if(NOT MW34_PAIRED_INCLUDE)
 message(FATAL_ERROR "Exact private paired pack metadata required")
endif()
add_executable(mw34_matherr_reference ../original.cpp "${WORKSPACE}/tools/m4-crt-control-integration/reference_x86.asm")
target_include_directories(mw34_matherr_reference PRIVATE "${MW34_PAIRED_INCLUDE}" "${CMAKE_CURRENT_SOURCE_DIR}/.." "${WORKSPACE}/tools/m4-crt")
target_compile_options(mw34_matherr_reference PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4 /WX /permissive- /showIncludes>)
target_link_libraries(mw34_matherr_reference PRIVATE bcrypt)
target_link_options(mw34_matherr_reference PRIVATE /BASE:0x60000000 /DYNAMICBASE:NO)
