cmake_minimum_required(VERSION 3.24)
project(MW34EntryControl LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "Windows AMD64 required")
endif()
get_filename_component(MW34_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
string(REPLACE "/EHsc" "" CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS}")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_executable(entry_control "${MW34_COMPOSITION_SOURCE}" "${MW34_CAPTURE_OBJECT}" "${MW34_ROOT}/tools/m4-crt/crt_services.cpp" "${MW34_ENTRY_OBJECT}" "${MW34_MATHERR_OBJECT}" "${MW34_CONTROL_OBJECT}" "${MW34_ROOT}/tools/m4-guest-cw/guest_cw.cpp")
target_include_directories(entry_control PRIVATE "${MW34_COMPOSITION_DIRECTORY}" "${CMAKE_CURRENT_SOURCE_DIR}/.." "${MW34_ROOT}/tools/m4-crt" "${MW34_ROOT}/tools/m4-crt-fdiv" "${MW34_ROOT}/tools/m4-runtime" "${MW34_ROOT}/tools/m4-startup" "${MW34_ROOT}/tools/m4-guest-cw" "${MW34_ROOT}/runtime")
target_include_directories(entry_control SYSTEM PRIVATE "${MW34_ROOT}/local/tools/remill-v6.0.1/remill-6.0.1/include")
target_compile_definitions(entry_control PRIVATE ADDRESS_SIZE_BITS=32 HAS_FEATURE_AVX=0 HAS_FEATURE_AVX512=0 MW34_ENTRY_OBJECT_SHA="${MW34_ENTRY_OBJECT_SHA}" MW34_CONTROL_OBJECT_SHA="${MW34_CONTROL_OBJECT_SHA}" MW34_REPLAY_INPUT_SHA="${MW34_REPLAY_INPUT_SHA}")
target_compile_options(entry_control PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/EHs;/showIncludes>)
target_link_libraries(entry_control PRIVATE "${MW34_FDIV_LIBRARY}" bcrypt)
