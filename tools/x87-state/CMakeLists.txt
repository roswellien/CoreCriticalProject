cmake_minimum_required(VERSION 3.24)
project(MW34X87State LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(MW34_X87_FIXTURE_KIND "0" CACHE STRING "Authored state:0push,1pop,2exchange,3free")
if(NOT MW34_X87_FIXTURE_KIND MATCHES "^[0-3]$")
 message(FATAL_ERROR "Unknown authored x87 state profile")
endif()
if(NOT WIN32)
 message(FATAL_ERROR "Hardware original x86 state reference requires Windows")
endif()
if(CMAKE_SIZEOF_VOID_P EQUAL 4)
 enable_language(ASM_MASM)
 add_executable(mw34_x87_reference reference_worker.cpp reference.asm)
 target_compile_definitions(mw34_x87_reference PRIVATE MW34_X87_FIXTURE_KIND=${MW34_X87_FIXTURE_KIND})
 target_compile_options(mw34_x87_reference PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4>)
 target_link_options(mw34_x87_reference PRIVATE /BASE:0x18000000 /DYNAMICBASE:NO /FIXED:NO)
else()
 if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang" OR NOT MW34_HOST_OBJECT OR NOT MW34_REMILL_INCLUDE)
  message(FATAL_ERROR "Native state worker requires Clang, static lifted object and pinned State")
 endif()
 add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../runtime" runtime)
 add_executable(mw34_x87_native native_worker.cpp "${MW34_HOST_OBJECT}")
 target_link_libraries(mw34_x87_native PRIVATE mw34_runtime)
 target_include_directories(mw34_x87_native SYSTEM PRIVATE "${MW34_REMILL_INCLUDE}")
 target_compile_definitions(mw34_x87_native PRIVATE MW34_X87_FIXTURE_KIND=${MW34_X87_FIXTURE_KIND} ADDRESS_SIZE_BITS=32 HAS_FEATURE_AVX=0 HAS_FEATURE_AVX512=0)
endif()
