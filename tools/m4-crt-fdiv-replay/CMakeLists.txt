cmake_minimum_required(VERSION 3.24)
project(MW34FDIVBeforeOnlyReplay LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR NOT MW34_FDIV_LIBRARY OR NOT MW34_REPLAY_INPUT_SHA)
 message(FATAL_ERROR "Frozen AMD64 Registry library and exact before-only input identity required")
endif()
add_executable(fdiv_replay replay.cpp)
target_include_directories(fdiv_replay PRIVATE ../m4-crt-fdiv ../m4-runtime ../m4-startup ../m4-guest-cw ../../runtime)
target_include_directories(fdiv_replay SYSTEM PRIVATE ../../local/tools/remill-v6.0.1/remill-6.0.1/include)
target_compile_definitions(fdiv_replay PRIVATE ADDRESS_SIZE_BITS=32 HAS_FEATURE_AVX=0 HAS_FEATURE_AVX512=0 MW34_REPLAY_INPUT_SHA="${MW34_REPLAY_INPUT_SHA}")
target_compile_options(fdiv_replay PRIVATE /EHsc)
target_link_libraries(fdiv_replay PRIVATE "${MW34_FDIV_LIBRARY}" bcrypt)
