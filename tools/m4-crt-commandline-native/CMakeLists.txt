cmake_minimum_required(VERSION 3.24)
project(MW34CommandLineNative LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR NOT MW34_CMD_INCLUDE OR NOT MW34_CMD_OBJECT)
  message(FATAL_ERROR "Exact Windows AMD64 typed command-line component and guest object required")
endif()
add_library(commandline_registry STATIC registry.cpp ../m4-runtime/runtime.cpp ../m4-startup/image_loader.cpp ../../runtime/memory/guest_memory.cpp "${MW34_CMD_OBJECT}")
target_include_directories(commandline_registry PUBLIC "${MW34_CMD_INCLUDE}" ../m4-runtime ../m4-startup ../../runtime)
target_include_directories(commandline_registry SYSTEM PUBLIC ../../local/tools/remill-v6.0.1/remill-6.0.1/include)
target_compile_definitions(commandline_registry PUBLIC ADDRESS_SIZE_BITS=32 HAS_FEATURE_AVX=0 HAS_FEATURE_AVX512=0)
target_compile_options(commandline_registry PRIVATE /EHsc -Wno-invalid-offsetof)
target_link_libraries(commandline_registry PUBLIC bcrypt)
add_executable(commandline_preflight preflight.cpp)
target_compile_options(commandline_preflight PRIVATE /EHsc -Wno-invalid-offsetof)
target_link_libraries(commandline_preflight PRIVATE commandline_registry)

add_executable(commandline_native native_worker.cpp)
target_compile_options(commandline_native PRIVATE /EHsc -Wno-invalid-offsetof)
target_link_libraries(commandline_native PRIVATE commandline_registry)
