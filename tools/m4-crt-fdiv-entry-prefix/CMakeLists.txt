cmake_minimum_required(VERSION 3.24)
project(MW34FDIVEntryPrefix LANGUAGES CXX)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "AMD64 native compiler required")
endif()
get_filename_component(MW34_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(fdiv_entry_prefix "${MW34_COMPOSITION_SOURCE}" "${MW34_ROOT}/tools/m4-crt/crt_services.cpp" "${MW34_ENTRY_OBJECT}")
target_include_directories(fdiv_entry_prefix PRIVATE "${MW34_COMPOSITION_DIRECTORY}" "${CMAKE_CURRENT_SOURCE_DIR}" "${MW34_ROOT}/tools/m4-crt" "${MW34_ROOT}/tools/m4-crt-fdiv" "${MW34_ROOT}/tools/m4-runtime" "${MW34_ROOT}/tools/m4-startup" "${MW34_ROOT}/tools/m4-guest-cw" "${MW34_ROOT}/runtime")
target_include_directories(fdiv_entry_prefix SYSTEM PRIVATE "${MW34_ROOT}/local/tools/remill-v6.0.1/remill-6.0.1/include")
target_compile_definitions(fdiv_entry_prefix PRIVATE ADDRESS_SIZE_BITS=32 HAS_FEATURE_AVX=0 HAS_FEATURE_AVX512=0 MW34_ENTRY_OBJECT_SHA="${MW34_ENTRY_OBJECT_SHA}" MW34_REPLAY_INPUT_SHA="${MW34_REPLAY_INPUT_SHA}")
target_compile_options(fdiv_entry_prefix PRIVATE /EHsc /showIncludes)
target_link_libraries(fdiv_entry_prefix PRIVATE "${MW34_FDIV_LIBRARY}" bcrypt)
