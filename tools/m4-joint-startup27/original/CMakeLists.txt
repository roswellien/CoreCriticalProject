cmake_minimum_required(VERSION 3.24)
project(MW34JointStartup27Original LANGUAGES CXX ASM_MASM)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4 OR NOT MW34_ORIGINAL_INCLUDE)
 message(FATAL_ERROR "Original27 requires x86 Windows and admitted private sparse pages")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(mw34_joint_startup27_original reference.cpp reference_x86.asm)
target_include_directories(mw34_joint_startup27_original PRIVATE "${MW34_ORIGINAL_INCLUDE}")
target_compile_options(mw34_joint_startup27_original PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4> $<$<COMPILE_LANGUAGE:CXX>:/WX> $<$<COMPILE_LANGUAGE:CXX>:/EHsc>)
target_link_libraries(mw34_joint_startup27_original PRIVATE bcrypt)
target_link_options(mw34_joint_startup27_original PRIVATE /BASE:0x18000000 /DYNAMICBASE:NO /FIXED:NO /LARGEADDRESSAWARE)
