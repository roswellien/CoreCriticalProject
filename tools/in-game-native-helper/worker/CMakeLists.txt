cmake_minimum_required(VERSION 3.24)
project(MW34LiveHelperWorker LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR NOT CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
  message(FATAL_ERROR "The helper worker requires Windows clang-cl targeting x64")
endif()
get_filename_component(WORKSPACE "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(FROZEN "${WORKSPACE}/reports/mw3-leaf-lift")
function(require_pin path wanted)
  if(NOT EXISTS "${path}")
    message(FATAL_ERROR "Missing reviewed helper input: ${path}")
  endif()
  file(SHA256 "${path}" actual)
  if(NOT actual STREQUAL wanted)
    message(FATAL_ERROR "Reviewed helper input identity differs: ${path}")
  endif()
endfunction()
require_pin("${FROZEN}/build.json" "3a51677200b38b30ec3500323ba4f937fdcfdcfc33f6c07ca7101add7359098e")
require_pin("${FROZEN}/layout.json" "a7d0c5084e1b7591384641b8605c7fcdade706178cb23b95d5d45c6bf3c496f4")
require_pin("${FROZEN}/sub_00409ed0.bin" "560c17c210dd7fedd7de601f42570c2a0473fa71962c13cfa85a005509a77c65")
require_pin("${WORKSPACE}/tools/lifting/lifted_worker.cpp" "6293c46c155103fb8ef6a288be39960dffe6defc2220a0ece22b5e5049e4c178")
require_pin("${WORKSPACE}/runtime/memory/guest_memory.cpp" "d21d9c166f7d165c422ce2c58442abebf94f1deacb6c40d5c9d9ba9e7a87ba11")
require_pin("${WORKSPACE}/runtime/memory/guest_memory.hpp" "cfb9b73c6ca0cd89d61ef304eaeef4434ae48ba27a49fe4173eac76d860f8d3b")
require_pin("${WORKSPACE}/runtime/core/guest_address.hpp" "87efd93a823bc0364ff7b989cbbc3ae6ba8e10fe747d5b0ecfe9e5daca62f4d9")
if(NOT MW34_HELPER_OBJECT AND MW34_LIFTED_OBJECT)
  set(MW34_HELPER_OBJECT "${MW34_LIFTED_OBJECT}")
endif()
if(NOT MW34_HELPER_OBJECT)
  message(FATAL_ERROR "MW34_HELPER_OBJECT must name the exact frozen helper O0 or O2 object")
endif()
file(REAL_PATH "${MW34_HELPER_OBJECT}" object_path)
foreach(level IN ITEMS O0 O2)
  file(REAL_PATH "${FROZEN}/sub_00409ed0-${level}.obj" candidate)
  if(object_path STREQUAL candidate)
    set(object_level "${level}")
  endif()
endforeach()
if(object_level STREQUAL "O0")
  set(object_sha "45ddad8ded561a01b87d193f1a1c183acf0f9db9796c5e8c60408480d2eb55e7")
elseif(object_level STREQUAL "O2")
  set(object_sha "23cde135338ac123fa7d067e5f618b57fe68677a2b684b45cbea8e3e5dd006e3")
else()
  message(FATAL_ERROR "Object must be the exact frozen helper path, not arbitrary native code")
endif()
require_pin("${object_path}" "${object_sha}")
add_executable(mw34_helper_worker "${CMAKE_CURRENT_SOURCE_DIR}/../worker.cpp" "${object_path}"
  "${WORKSPACE}/runtime/memory/guest_memory.cpp")
target_include_directories(mw34_helper_worker PRIVATE "${WORKSPACE}/runtime" "${WORKSPACE}/tools/m4-startup")
target_include_directories(mw34_helper_worker SYSTEM PRIVATE "${WORKSPACE}/local/tools/remill-v6.0.1/remill-6.0.1/include")
target_compile_options(mw34_helper_worker PRIVATE /W4 /WX /EHsc /permissive-)
target_compile_definitions(mw34_helper_worker PRIVATE ADDRESS_SIZE_BITS=32 HAS_FEATURE_AVX=0 HAS_FEATURE_AVX512=0
  MW34_HELPER_OBJECT_SHA="${object_sha}" MW34_HELPER_OBJECT_PATH="${object_path}"
  MW34_HELPER_BUILD_PATH="${FROZEN}/build.json" MW34_HELPER_ADAPTOR_PATH="${WORKSPACE}/tools/lifting/lifted_worker.cpp")
target_link_libraries(mw34_helper_worker PRIVATE bcrypt)
