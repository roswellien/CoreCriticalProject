cmake_minimum_required(VERSION 3.24)
project(MW34M4CRT LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(MW34_WORKSPACE "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_subdirectory("${MW34_WORKSPACE}/tools/m4-runtime" runtime)
add_library(mw34_m4_crt STATIC crt_services.cpp)
target_link_libraries(mw34_m4_crt PUBLIC mw34_m4_runtime)
target_include_directories(mw34_m4_crt PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(mw34_m4_crt_tests "${MW34_WORKSPACE}/tests/m4-crt/crt_tests.cpp")
target_link_libraries(mw34_m4_crt_tests PRIVATE mw34_m4_crt)
target_compile_options(mw34_m4_crt PRIVATE /W4 /WX /permissive-)
target_compile_options(mw34_m4_crt_tests PRIVATE /W4 /WX /permissive-)
enable_testing()
add_test(NAME m4_crt_contract COMMAND mw34_m4_crt_tests "${MW34_WORKSPACE}/local/mw3-staged/Shared_DLLs/MSVCRT.DLL")
if(MW34_CRT_REPLAY_INCLUDE)
 add_library(mw34_m4_crt_O2 ALIAS mw34_m4_crt)
 add_library(mw34_m4_crt_O0 STATIC crt_services.cpp)
 target_link_libraries(mw34_m4_crt_O0 PUBLIC mw34_m4_runtime)
 target_include_directories(mw34_m4_crt_O0 PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
 target_compile_options(mw34_m4_crt_O0 PRIVATE /W4 /WX /permissive- /Od)
 foreach(level O0 O2)
  add_executable(mw34_m4_crt_replay_${level} native_replay.cpp)
  target_include_directories(mw34_m4_crt_replay_${level} PRIVATE "${MW34_CRT_REPLAY_INCLUDE}")
  target_include_directories(mw34_m4_crt_replay_${level} SYSTEM PRIVATE "${MW34_WORKSPACE}/local/tools/remill-v6.0.1/remill-6.0.1/include")
  target_compile_definitions(mw34_m4_crt_replay_${level} PRIVATE ADDRESS_SIZE_BITS=32 HAS_FEATURE_AVX=0 HAS_FEATURE_AVX512=0)
  target_link_libraries(mw34_m4_crt_replay_${level} PRIVATE mw34_m4_crt_${level})
  target_compile_options(mw34_m4_crt_replay_${level} PRIVATE /W4 /WX /permissive-)
  if(level STREQUAL "O0")
   target_compile_options(mw34_m4_crt_replay_${level} PRIVATE /Od)
  endif()
 endforeach()
endif()
