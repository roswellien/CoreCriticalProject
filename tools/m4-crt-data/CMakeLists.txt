cmake_minimum_required(VERSION 3.24)
project(MW34CRTCommandLineOriginal LANGUAGES CXX ASM_MASM)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT MW34_COMMANDLINE_INCLUDE)
 message(FATAL_ERROR "Exact private command-line fragment metadata required")
endif()
add_executable(mw34_commandline_reference commandline_reference.cpp commandline_x86.asm)
target_include_directories(mw34_commandline_reference PRIVATE "${MW34_COMMANDLINE_INCLUDE}" "${CMAKE_CURRENT_SOURCE_DIR}/../m4-crt")
target_compile_options(mw34_commandline_reference PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4 /WX /permissive->)
target_link_libraries(mw34_commandline_reference PRIVATE bcrypt)
