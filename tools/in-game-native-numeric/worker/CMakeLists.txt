cmake_minimum_required(VERSION 3.24)
project(MW34LiveNumericWorker LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR NOT CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
  message(FATAL_ERROR "The numeric worker requires clang-cl targeting x64")
endif()
get_filename_component(WORKSPACE "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(FROZEN "${WORKSPACE}/reports/m4-fp-numeric-max-carry")
set(ANCESTOR_DIR "${FROZEN}/native-preparation-canonical-r1")
set(SESSION_DIR "${WORKSPACE}/reports/mw3-live-native-numeric/session-mxcsr-r1")
set(GATE "${FROZEN}/root-component-acceptance-r2/report.json")
function(require_pin path wanted)
  if(NOT EXISTS "${path}")
    message(FATAL_ERROR "Missing frozen input: ${path}")
  endif()
  file(SHA256 "${path}" actual)
  if(NOT actual STREQUAL wanted)
    message(FATAL_ERROR "Frozen input identity differs: ${path}")
  endif()
endfunction()
require_pin("${ANCESTOR_DIR}/session.hpp" "22bef4a73a171b9900b0376d7d2c3022d208b5e8b2c6159ccad2d060834da9b1")
require_pin("${SESSION_DIR}/session.hpp" "c864029d2758aadef19e2604a775775372cd4c7077cc358cc914cdf529db45ea")
file(READ "${ANCESTOR_DIR}/session.hpp" ancestor_session)
file(READ "${SESSION_DIR}/session.hpp" derivative_session)
string(REPLACE "s_.x87.fxsave.mxcsr.flat!=0x1f80"
  "(s_.x87.fxsave.mxcsr.flat!=0x1f80&&s_.x87.fxsave.mxcsr.flat!=0x1fa0)"
  sole_delta_session "${ancestor_session}")
if(NOT derivative_session STREQUAL sole_delta_session)
  message(FATAL_ERROR "Private Session must differ only by the exact MXCSR1f80/1fa0 admission clause")
endif()
require_pin("${ANCESTOR_DIR}/native.cpp" "3fa951459f194481e9f267d43525f564f8f94b251be5d7273269a0bf035ec326")
require_pin("${GATE}" "fe713df4600d61acb3ac1eb6ea80963c1dc225d5e73fa3032795dc1905ca7706")
if(NOT MW34_NMX_OBJECT)
  message(FATAL_ERROR "MW34_NMX_OBJECT must name the exact frozen canonical native-O0.obj or native-O2.obj")
endif()
file(REAL_PATH "${MW34_NMX_OBJECT}" object_path)
foreach(level IN ITEMS O0 O2)
  file(REAL_PATH "${FROZEN}/native-canonical-r1/native-${level}.obj" candidate)
  if(object_path STREQUAL candidate)
    set(object_level "${level}")
  endif()
endforeach()
if(object_level STREQUAL "O0")
  set(object_sha "1a83321f462a0d235036fecb542a6ddaf859956302b493ffa80e7320b7d3b25c")
elseif(object_level STREQUAL "O2")
  set(object_sha "7695ec1853c53bcc4e4c868de264f7145ef6d5fc44edde6a0e4f77722871e296")
else()
  message(FATAL_ERROR "Object must be the frozen canonical path, not an arbitrary native object")
endif()
require_pin("${object_path}" "${object_sha}")
string(REPLACE "/EHsc" "" CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS}")
add_executable(mw34_numeric_worker "${CMAKE_CURRENT_SOURCE_DIR}/../worker.cpp" "${object_path}"
  "${WORKSPACE}/tools/m4-runtime/runtime.cpp"
  "${WORKSPACE}/tools/m4-startup/image_loader.cpp"
  "${WORKSPACE}/runtime/memory/guest_memory.cpp")
target_include_directories(mw34_numeric_worker PRIVATE "${SESSION_DIR}" "${WORKSPACE}/tools/m4-runtime"
  "${WORKSPACE}/tools/m4-startup" "${WORKSPACE}/runtime")
target_include_directories(mw34_numeric_worker SYSTEM PRIVATE "${WORKSPACE}/local/tools/remill-v6.0.1/remill-6.0.1/include")
target_compile_options(mw34_numeric_worker PRIVATE /W4 /WX /EHs /permissive-)
target_compile_definitions(mw34_numeric_worker PRIVATE ADDRESS_SIZE_BITS=32 HAS_FEATURE_AVX=0 HAS_FEATURE_AVX512=0
  MW34_NMX_OBJECT_SHA="${object_sha}" MW34_NMX_OBJECT_PATH="${object_path}"
  MW34_NMX_SESSION_PATH="${SESSION_DIR}/session.hpp" MW34_NMX_ANCESTOR_PATH="${ANCESTOR_DIR}/session.hpp"
  MW34_NMX_GATE_PATH="${GATE}")
target_link_libraries(mw34_numeric_worker PRIVATE bcrypt)
