cmake_minimum_required(VERSION 3.24)
project(MW34LiveNumericX86 LANGUAGES CXX ASM_MASM)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "The live numeric bridge requires Win32 MSVC")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(mw34_numeric_hook SHARED ../hook.cpp ../numeric_x86.asm ../exports.def)
target_link_libraries(mw34_numeric_hook PRIVATE bcrypt)
add_executable(mw34_numeric_launcher ../launcher.cpp)
target_link_libraries(mw34_numeric_launcher PRIVATE bcrypt)
foreach(target mw34_numeric_hook mw34_numeric_launcher)
    target_compile_definitions(${target} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_compile_options(${target} PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4 /WX /EHsc>)
endforeach()
