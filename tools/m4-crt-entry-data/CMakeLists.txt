cmake_minimum_required(VERSION 3.24)
project(MW34EntryData LANGUAGES CXX ASM)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "Windows AMD64 required")
endif()
get_filename_component(MW34_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_executable(entry_data "${MW34_COMPOSITION_SOURCE}" capture.s "${MW34_ROOT}/tools/m4-crt/crt_services.cpp" "${MW34_ENTRY_OBJECT}")
target_include_directories(entry_data PRIVATE "${MW34_COMPOSITION_DIRECTORY}" . "${MW34_ROOT}/tools/m4-crt" "${MW34_ROOT}/tools/m4-crt-fdiv" "${MW34_ROOT}/tools/m4-runtime" "${MW34_ROOT}/tools/m4-startup" "${MW34_ROOT}/tools/m4-guest-cw" "${MW34_ROOT}/runtime")
target_include_directories(entry_data SYSTEM PRIVATE "${MW34_ROOT}/local/tools/remill-v6.0.1/remill-6.0.1/include")
target_compile_definitions(entry_data PRIVATE ADDRESS_SIZE_BITS=32 HAS_FEATURE_AVX=0 HAS_FEATURE_AVX512=0 MW34_ENTRY_OBJECT_SHA="${MW34_ENTRY_OBJECT_SHA}" MW34_REPLAY_INPUT_SHA="${MW34_REPLAY_INPUT_SHA}")
target_compile_options(entry_data PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/EHsc;/showIncludes>)
target_link_libraries(entry_data PRIVATE "${MW34_FDIV_LIBRARY}" bcrypt)
