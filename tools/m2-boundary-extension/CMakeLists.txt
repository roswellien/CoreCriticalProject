cmake_minimum_required(VERSION 3.24)
project(MW34M2BoundaryExtension LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(CMAKE_SIZEOF_VOID_P EQUAL 4)
 enable_language(ASM_MASM)
 add_executable(m2_reference reference.cpp reference.asm)
 target_link_libraries(m2_reference PRIVATE bcrypt winmm)
 target_link_options(m2_reference PRIVATE /BASE:0x18000000 /DYNAMICBASE:NO /FIXED:NO "/MAP:${CMAKE_BINARY_DIR}/m2_reference.map")
else()
 if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang" OR NOT M2_OBJECTS OR NOT MW34_REMILL_INCLUDE)
  message(FATAL_ERROR "Native worker requires Clang, pinned State headers and generated objects")
 endif()
 add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../runtime" runtime)
 add_executable(m2_native native.cpp ${M2_OBJECTS})
 target_compile_options(m2_native PRIVATE /EHs /EHc-)
 target_link_libraries(m2_native PRIVATE mw34_runtime winmm bcrypt)
 target_include_directories(m2_native SYSTEM PRIVATE "${MW34_REMILL_INCLUDE}")
 target_compile_definitions(m2_native PRIVATE ADDRESS_SIZE_BITS=32 HAS_FEATURE_AVX=0 HAS_FEATURE_AVX512=0)
endif()
