cmake_minimum_required(VERSION 3.24)
project(MW34NumericMax LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32 OR NOT MW34_PROBE_INCLUDE)
 message(FATAL_ERROR "Windows exact extraction required")
endif()
if(CMAKE_SIZEOF_VOID_P EQUAL 4)
 enable_language(ASM_MASM)
 add_executable(mw34_numeric_reference reference.cpp reference_x86.asm)
 target_include_directories(mw34_numeric_reference PRIVATE "${MW34_PROBE_INCLUDE}")
 target_compile_options(mw34_numeric_reference PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/EHsc>)
 target_link_libraries(mw34_numeric_reference PRIVATE bcrypt)
 target_link_options(mw34_numeric_reference PRIVATE /BASE:0x18000000 /DYNAMICBASE:NO /FIXED:NO /LARGEADDRESSAWARE)
else()
 if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang" OR NOT MW34_HOST_OBJECT OR NOT MW34_REMILL_INCLUDE)
  message(FATAL_ERROR "Clang and exact objects/State required")
 endif()
 add_executable(mw34_numeric_native native.cpp ../../runtime/memory/guest_memory.cpp "${MW34_HOST_OBJECT}")
 target_include_directories(mw34_numeric_native PRIVATE ../../runtime "${MW34_PROBE_INCLUDE}")
 target_include_directories(mw34_numeric_native SYSTEM PRIVATE "${MW34_REMILL_INCLUDE}")
 target_compile_definitions(mw34_numeric_native PRIVATE ADDRESS_SIZE_BITS=32 HAS_FEATURE_AVX=0 HAS_FEATURE_AVX512=0 MW34_NUMERIC_REPAIRED=$<BOOL:${MW34_REPAIRED}>)
 target_compile_options(mw34_numeric_native PRIVATE /EHsc /clang:-frounding-math)
 target_link_libraries(mw34_numeric_native PRIVATE bcrypt)
endif()
