cmake_minimum_required(VERSION 3.24)
project(MW34FDIVNative LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR NOT MW34_FDIV_INCLUDE OR NOT MW34_FDIV_OBJECT)
  message(FATAL_ERROR "Exact Windows AMD64 registry and private static guest object required")
endif()
add_library(fdiv_registry STATIC registry.cpp ../m4-runtime/runtime.cpp ../m4-startup/image_loader.cpp
  ../m4-guest-cw/guest_cw.cpp ../../runtime/memory/guest_memory.cpp "${MW34_FDIV_OBJECT}")
target_include_directories(fdiv_registry PUBLIC "${MW34_FDIV_INCLUDE}" . ../m4-runtime ../m4-startup ../m4-guest-cw ../../runtime)
target_include_directories(fdiv_registry SYSTEM PUBLIC ../../local/tools/remill-v6.0.1/remill-6.0.1/include)
target_compile_definitions(fdiv_registry PUBLIC ADDRESS_SIZE_BITS=32 HAS_FEATURE_AVX=0 HAS_FEATURE_AVX512=0)
target_compile_options(fdiv_registry PRIVATE /EHsc)
target_link_libraries(fdiv_registry PUBLIC bcrypt)
add_executable(fdiv_native_tests ../../tests/m4-crt-fdiv/native_tests.cpp)
target_compile_options(fdiv_native_tests PRIVATE /EHsc)
target_link_libraries(fdiv_native_tests PRIVATE fdiv_registry)
