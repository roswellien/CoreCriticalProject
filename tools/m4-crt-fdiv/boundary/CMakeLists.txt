cmake_minimum_required(VERSION 3.24)
project(MW34FDIVBoundary LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(fdiv_boundary boundary.cpp "${MW34_BOUNDARY_OBJECT}")
target_include_directories(fdiv_boundary PRIVATE .. ../../m4-runtime ../../m4-startup ../../m4-guest-cw ../../../runtime)
target_include_directories(fdiv_boundary SYSTEM PRIVATE ../../../local/tools/remill-v6.0.1/remill-6.0.1/include)
target_compile_definitions(fdiv_boundary PRIVATE ADDRESS_SIZE_BITS=32 HAS_FEATURE_AVX=0 HAS_FEATURE_AVX512=0)
target_compile_options(fdiv_boundary PRIVATE /EHsc)
target_link_libraries(fdiv_boundary PRIVATE "${MW34_FDIV_LIBRARY}" bcrypt)
